Missing methods in GlueBackend causing AttributeError

#### Description

Several Glue operations are failing with AttributeError because the corresponding methods have been removed from the GlueBackend class. When trying to use `stop_trigger`, `stop_session`, or `get_connection`, the client raises an AttributeError indicating the methods don't exist.

#### Steps/Code to Reproduce

```python
import boto3
from moto import mock_aws

@mock_aws
def test_stop_trigger_error():
    client = boto3.client('glue', region_name='us-east-1')
    # Create a trigger first
    client.create_trigger(
        Name='test-trigger',
        Type='SCHEDULED',
        Schedule='cron(5 3 * * ? *)',
        Actions=[{'JobName': 'test-job'}],
        StartOnCreation=True
    )
    
    # This will fail
    client.stop_trigger(Name='test-trigger')

@mock_aws  
def test_get_connection_error():
    client = boto3.client('glue', region_name='us-east-1')
    # Create a connection first
    client.create_connection(
        ConnectionInput={
            'Name': 'test-connection',
            'ConnectionType': 'JDBC',
            'ConnectionProperties': {'key': 'value'}
        }
    )
    
    # This will fail
    client.get_connection(Name='test-connection')

@mock_aws
def test_stop_session_error():
    client = boto3.client('glue', region_name='us-east-1')
    # Create a session first
    resp = client.create_session(
        Id='test-session',
        Role='test-role'
    )
    
    # This will fail
    client.stop_session(Id='test-session')
```
