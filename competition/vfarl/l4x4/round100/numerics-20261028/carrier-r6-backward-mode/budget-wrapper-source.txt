"""Linux budget supervisor for the frozen one-backward mode-study payload.

This is the readable control source embedded in run.py. The generated carrier
adds only its inert EMBEDDED_PAYLOAD literal and the __main__ invocation.
"""
from pathlib import Path
import base64
import ctypes
import hashlib
import json
import os
import signal
import subprocess
import sys
import time
import zlib


FROZEN_PAYLOAD_SHA256 = 'e46de57695e84807126dccd14b8a81a9992a607ca505681f922c3d46d01e7556'
EXECUTION_SECONDS = 1180.0
CLEANUP_SECONDS = 20.0
PR_SET_CHILD_SUBREAPER = 36
PR_GET_CHILD_SUBREAPER = 37


def proc_identity(pid):
    """Read Linux stat without assuming the comm field contains no spaces/')."""
    text = Path('/proc', str(pid), 'stat').read_text()
    left, separator, tail = text.rpartition(')')
    if not separator:
        raise RuntimeError('Malformed Linux process stat')
    fields = tail.split()
    actual_pid = int(left.split('(', 1)[0].strip())
    if actual_pid != int(pid) or len(fields) < 20:
        raise RuntimeError('Incomplete Linux process identity')
    return {'pid': actual_pid, 'ppid': int(fields[1]),
            'starttime': int(fields[19]), 'state': fields[0]}


def proc_table():
    table = {}
    for path in Path('/proc').iterdir():
        if path.name.isdigit():
            try:
                row = proc_identity(int(path.name))
            except (FileNotFoundError, ProcessLookupError, PermissionError):
                continue
            table[row['pid']] = row
    return table


def enable_subreaper():
    if sys.platform != 'linux' or not Path('/proc/self/stat').is_file():
        raise RuntimeError('Linux prctl and procfs are required before payload execution')
    if os.getpid() <= 1:
        raise RuntimeError('Namespace init cannot safely supervise only this payload')
    libc = ctypes.CDLL(None, use_errno=True)
    prctl = libc.prctl
    prctl.argtypes = [ctypes.c_int, ctypes.c_ulong, ctypes.c_ulong,
                      ctypes.c_ulong, ctypes.c_ulong]
    prctl.restype = ctypes.c_int
    if prctl(PR_SET_CHILD_SUBREAPER, 1, 0, 0, 0) != 0:
        raise OSError(ctypes.get_errno(), 'PR_SET_CHILD_SUBREAPER failed')
    enabled = ctypes.c_int(0)
    address = ctypes.cast(ctypes.pointer(enabled), ctypes.c_void_p).value
    if prctl(PR_GET_CHILD_SUBREAPER, address, 0, 0, 0) != 0 or enabled.value != 1:
        raise RuntimeError('Child subreaper setup could not be verified')
    if not hasattr(os, 'pidfd_open') or not hasattr(signal, 'pidfd_send_signal'):
        raise RuntimeError('Linux pidfd signalling is required before payload execution')
    descriptor = os.pidfd_open(os.getpid(), 0)
    try:
        signal.pidfd_send_signal(descriptor, 0, None, 0)
    finally:
        os.close(descriptor)
    return proc_identity(os.getpid())


def discover_owned(wrapper, owned, excluded):
    """Only wrapper descendants, including orphan adoption; never a PGID scan."""
    table = proc_table()
    actual = table.get(wrapper['pid'])
    if actual is None or actual['starttime'] != wrapper['starttime']:
        raise RuntimeError('Supervisor process identity changed')
    changed = True
    while changed:
        changed = False
        for row in table.values():
            pid, start = row['pid'], row['starttime']
            if pid == wrapper['pid'] or (pid, start) in excluded:
                continue
            if row['starttime'] < wrapper['starttime']:
                continue
            parent = table.get(row['ppid'])
            direct = row['ppid'] == wrapper['pid']
            descendant = bool(parent and owned.get(parent['pid']) == parent['starttime'])
            if (direct or descendant) and owned.get(pid) != start:
                owned[pid] = start
                changed = True
    return {pid: row for pid, row in table.items()
            if pid != wrapper['pid'] and owned.get(pid) == row['starttime']}


def signal_owned(pid, starttime, sig, wrapper_pid):
    """Re-read starttime immediately before signalling a previously owned PID."""
    if pid == wrapper_pid or pid <= 1:
        raise RuntimeError('Refusing an invalid supervisor signal target')
    try:
        descriptor = os.pidfd_open(pid, 0)
    except ProcessLookupError:
        return False
    try:
        try:
            actual = proc_identity(pid)
        except (FileNotFoundError, ProcessLookupError):
            return False
        if actual['starttime'] != starttime or actual['state'] == 'Z':
            return False
        try:
            signal.pidfd_send_signal(descriptor, sig, None, 0)
        except ProcessLookupError:
            return False
        return True
    finally:
        os.close(descriptor)


def write_receipt(path, receipt):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    temporary = path.with_name(path.name + '.part-' + str(os.getpid()))
    with temporary.open('w', encoding='utf-8') as stream:
        json.dump(receipt, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())
    temporary.replace(path)


def reap_owned(child, owned, reaped):
    # Preserve Popen's actual direct-child returncode before any other waitpid.
    code = child.poll() if child is not None else None
    for pid, start in list(owned.items()):
        if child is not None and pid == child.pid:
            continue
        try:
            identity = proc_identity(pid)
        except (FileNotFoundError, ProcessLookupError):
            continue
        if identity['starttime'] != start:
            continue
        try:
            actual, status = os.waitpid(pid, os.WNOHANG)
        except ChildProcessError:
            continue
        if actual:
            reaped.append({'pid': actual, 'starttime': start, 'wait_status': status})
    return code


def cleanup_owned(wrapper, child, owned, excluded, deadline):
    started = time.monotonic()
    # TERM has at most five seconds; KILL/reaping uses the remaining budget.
    kill_at = min(started + 5.0, started + max(0.0, deadline - started) / 2.0)
    sent_term, sent_kill, reaped, errors = set(), set(), [], []
    live = {}
    while True:
        try:
            live = discover_owned(wrapper, owned, excluded)
            reap_owned(child, owned, reaped)
            live = discover_owned(wrapper, owned, excluded)
        except Exception as error:
            errors.append(type(error).__name__ + ': ' + str(error))
            break
        if not live:
            break
        now = time.monotonic()
        sig = signal.SIGKILL if now >= kill_at else signal.SIGTERM
        sent = sent_kill if sig == signal.SIGKILL else sent_term
        for pid, row in live.items():
            key = (pid, row['starttime'])
            if key not in sent:
                try:
                    if signal_owned(pid, row['starttime'], sig, wrapper['pid']):
                        sent.add(key)
                except Exception as error:
                    errors.append(type(error).__name__ + ': ' + str(error))
        reap_owned(child, owned, reaped)
        if time.monotonic() >= deadline:
            try:
                live = discover_owned(wrapper, owned, excluded)
            except Exception as error:
                errors.append(type(error).__name__ + ': ' + str(error))
            break
        time.sleep(min(0.02, max(0.0, deadline - time.monotonic())))
    return {'cleanup_started_monotonic': started,
            'cleanup_finished_monotonic': time.monotonic(),
            'cleanup_deadline_monotonic': deadline,
            'term_sent': [{'pid': p, 'starttime': s} for p, s in sorted(sent_term)],
            'kill_sent': [{'pid': p, 'starttime': s} for p, s in sorted(sent_kill)],
            'adopted_reaped': reaped, 'survivors': list(live.values()),
            'errors': errors, 'cleanup_confirmed': not live and not errors}


def run_bounded_command(command, receipt_path, execution_seconds=EXECUTION_SECONDS,
                        cleanup_seconds=CLEANUP_SECONDS, started=None, provenance=None):
    """Runtime primitive; the GPU entry always supplies the frozen constants."""
    started = time.monotonic() if started is None else started
    if execution_seconds <= 0 or cleanup_seconds <= 0:
        raise ValueError('Execution and cleanup budgets must be positive')
    deadline = started + execution_seconds
    final_deadline = deadline + cleanup_seconds
    receipt = {'format': 'vfarl-linux-bounded-payload-receipt-v1',
               'wrapper_pid': os.getpid(), 'started_monotonic': started,
               'execution_deadline_monotonic': deadline,
               'hard_cleanup_deadline_monotonic': final_deadline,
               'execution_seconds': execution_seconds, 'cleanup_seconds': cleanup_seconds,
               'payload_started': False, 'status': 'preflight', 'provenance': provenance,
               'study_completion_inferred': False, 'adapter_restore_inferred': False}
    child, wrapper, owned, excluded = None, None, {}, set()
    interrupted, previous_handlers = [], {}
    reason, failure, code = 'preflight_failure', None, None
    try:
        wrapper = enable_subreaper()
        # This fresh supervisor must own no unrelated pre-existing child tree.
        excluded = {(r['pid'], r['starttime']) for r in proc_table().values()
                    if r['ppid'] == wrapper['pid']}
        if excluded:
            raise RuntimeError('Fresh budget supervisor already has child processes')
        receipt['wrapper_identity'] = wrapper
        receipt['prctl_subreaper_verified'] = True
        write_receipt(receipt_path, receipt)
        for signum in (signal.SIGTERM, signal.SIGINT, signal.SIGHUP):
            previous_handlers[signum] = signal.getsignal(signum)
            signal.signal(signum, lambda sig, frame: interrupted.append(sig))
        if time.monotonic() >= deadline:
            reason = 'execution_budget_exhausted_before_payload'
        else:
            # stdout/stderr/cwd/environment are inherited, with no result filtering.
            child = subprocess.Popen(command, start_new_session=True)
            receipt.update(payload_started=True, child_pid=child.pid, status='running')
            write_receipt(receipt_path, receipt)
            reason = 'running'
            while True:
                code = child.poll()
                discover_owned(wrapper, owned, excluded)
                if code is not None:
                    reason = 'child_exited'
                    break
                if interrupted:
                    reason = 'supervisor_signal'
                    break
                if time.monotonic() >= deadline:
                    reason = 'execution_timeout'
                    break
                time.sleep(min(0.02, max(0.0, deadline - time.monotonic())))
    except Exception as error:
        failure = type(error).__name__ + ': ' + str(error)
        reason = 'supervisor_error' if child is not None else 'preflight_failure'
    finally:
        if wrapper is not None and child is not None:
            # Keep part of the same cleanup budget for receipt/exit bookkeeping.
            reserve = min(0.5, cleanup_seconds / 10.0)
            cleanup_deadline = min(final_deadline - reserve,
                                   time.monotonic() + cleanup_seconds - reserve)
            try:
                cleanup = cleanup_owned(wrapper, child, owned, excluded, cleanup_deadline)
            except Exception as error:
                cleanup = {'cleanup_confirmed': False, 'survivors': [],
                           'survivor_state_unverified': True,
                           'errors': [type(error).__name__ + ': ' + str(error)]}
            code = child.poll()
        else:
            cleanup = {'cleanup_confirmed': child is None, 'survivors': [],
                       'errors': [], 'payload_was_not_started': child is None}
        for signum, handler in previous_handlers.items():
            signal.signal(signum, handler)
        receipt.update(status=reason, failure=failure, child_returncode=code,
                       received_signals=interrupted, cleanup=cleanup,
                       finished_monotonic=time.monotonic())
        receipt['elapsed_seconds'] = receipt['finished_monotonic'] - started
        receipt['within_overall_budget'] = receipt['finished_monotonic'] <= final_deadline
        try:
            write_receipt(receipt_path, receipt)
        except Exception as error:
            receipt['receipt_write_error'] = type(error).__name__ + ': ' + str(error)
            print(json.dumps(receipt, ensure_ascii=False, allow_nan=False),
                  file=sys.stderr, flush=True)
    if receipt.get('receipt_write_error'):
        return 125
    if not cleanup['cleanup_confirmed'] or not receipt['within_overall_budget']:
        return 125
    if reason in {'execution_timeout', 'execution_budget_exhausted_before_payload'}:
        return 124
    if reason == 'supervisor_signal':
        return 128 + interrupted[0]
    if reason != 'child_exited':
        return 125
    return code if code is not None and code >= 0 else 128 + abs(code or 0)


def main():
    started = time.monotonic()
    working = Path('/kaggle/working')
    receipt_path = working / 'vfarl-budget-receipt.json'
    try:
        payload = zlib.decompress(base64.b64decode(EMBEDDED_PAYLOAD, validate=True))
        if hashlib.sha256(payload).hexdigest() != FROZEN_PAYLOAD_SHA256:
            raise RuntimeError('Frozen payload SHA256 mismatch before execution')
        working.mkdir(parents=True, exist_ok=True)
        payload_path = working / 'vfarl-frozen-backward-mode-r6-payload.py'
        with payload_path.open('wb') as stream:
            stream.write(payload)
        if hashlib.sha256(payload_path.read_bytes()).hexdigest() != FROZEN_PAYLOAD_SHA256:
            raise RuntimeError('Materialized payload SHA256 mismatch before execution')
    except Exception as error:
        write_receipt(receipt_path, {'format': 'vfarl-linux-bounded-payload-receipt-v1',
            'status': 'payload_integrity_failure', 'payload_started': False,
            'failure': type(error).__name__ + ': ' + str(error),
            'elapsed_seconds': time.monotonic() - started,
            'study_completion_inferred': False, 'adapter_restore_inferred': False})
        return 125
    return run_bounded_command([sys.executable, str(payload_path)], receipt_path,
        execution_seconds=EXECUTION_SECONDS, cleanup_seconds=CLEANUP_SECONDS,
        started=started, provenance={'frozen_payload_sha256': FROZEN_PAYLOAD_SHA256,
                                    'payload_path': str(payload_path)})
