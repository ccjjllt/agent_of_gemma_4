```python
from flake8.style_guide import Decision, DecisionEngine, Ignored, Selected

def options(select, defaults):
    return Namespace(
        select=select,
        ignore=[],
        extend_select=None,
        extend_ignore=None,
        extended_default_select=defaults,
        extended_default_ignore=[],
    )

assert explicit.was_selected("E111") is Selected.Explicitly
assert explicit.was_selected("E112") is Ignored.Implicitly
assert explicit.decision_for("E111") is Decision.Selected
assert explicit.decision_for("E112") is Decision.Ignored

implicit = DecisionEngine(options(None, ["E"]))
assert implicit.was_selected("E112") is Selected.Implicitly
```
