# CELL 0: markdown
# robust-edit: reliable file edits for the Gemma 4 agent harness (31B A/B)

**robust-edit** is a drop-in skill (dataset: `magicsword001/robust-edit`, Apache 2.0) that removes editing failures
caused by the harness rather than by the model:

- `edit_file` often fails on 31B: a string argument gets closed with a backtick and the other arguments are lost,
  or `old_string` does not match.
- Tool results reach the model as JSON, so code shows `\n` and `\"`; models copy these escapes into
  `old_string`, and `edit_file`'s fallback does not undo a backslash followed by a real line break.
- After a failed edit the model tends to repeat the identical call.
- The harness's reset of agent-edited test files is a no-op in swegemma 0.2.7, so one edited test file makes the
  hidden test patch fail to apply.

A one-call setup step installs a small editor into `.git/agent/edit.py` and keeps test-file edits out of the patch.
The agent then edits with one `run_command` (SEARCH/REPLACE blocks in a heredoc); the editor undoes copied escapes,
never writes a Python file that does not compile, and answers a miss with the closest lines.

This notebook (1) demonstrates the editor on a toy repository (no GPU needed), then (2) runs Gemma 4 31B on 4×L4
with two agents that are identical except for robust-edit, alternately on the same training tasks, and counts
editing outcomes. Traces, patches and logs are deleted at the end; only the counts are kept.

import glob, os, shutil, subprocess, sys, tempfile
from pathlib import Path

SKILL_DIR = next(Path(p).parent for p in sorted(glob.glob('/kaggle/input/**/skills/robust-edit/SKILL.md', recursive=True))
                 if (Path(p).parents[2] / 'README.md').is_file() and (Path(p).parents[2] / 'example').is_dir())
EXAMPLE_DIR = SKILL_DIR.parents[1] / 'example'
print('skill:', SKILL_DIR)

# A toy repository with the editor installed the way setup.py does it
demo = Path(tempfile.mkdtemp())
(demo / 'pkg').mkdir()
(demo / 'pkg' / 'mod.py').write_text('def greet(name):\n    if not name:\n        return ""\n    return f"hi {name}"\n')
subprocess.run(['git', 'init', '-q'], cwd=demo, check=True)
(demo / '.git' / 'agent').mkdir()
shutil.copyfile(SKILL_DIR / 'scripts' / 'edit_tool.py', demo / '.git' / 'agent' / 'edit.py')


def edit(args, stdin=''):
    out = subprocess.run([sys.executable, '.git/agent/edit.py', *args], cwd=demo, input=stdin,
                         capture_output=True, text=True).stdout
    print(f'$ python3 .git/agent/edit.py {" ".join(args)}\n{out}')


block = lambda s, r: f'<<<<<<< SEARCH\n{s}\n=======\n{r}\n>>>>>>> REPLACE\n'
# 1. SEARCH copied with JSON escapes from a tool result (backslash + line break, \")
edit(['pkg/mod.py'], block('    if not name:\\\n        return \\"\\"', '    if not name:\\\n        return "nobody"'))
# 2. An edit that would break the syntax is refused
edit(['pkg/mod.py'], block('    if not name:', '    if not name'))
# 3. SEARCH that is not in the file: the closest lines come back with line numbers
edit(['pkg/mod.py'], block('    return f"hello {name}"', '    return f"hello, {name}"'))

# CELL 10: code
import asyncio, concurrent.futures, time
import pandas as pd
from google.adk.agents.context_cache_config import ContextCacheConfig
from google.adk.apps._configs import EventsCompactionConfig
from swegemma.config import EvalConfig, build_submission_limits
from swegemma.evaluate import Evaluator


def run_sync(coro_or_fn, *args, **kwargs):
    fn = (lambda: coro_or_fn(*args, **kwargs)) if callable(coro_or_fn) else (lambda: coro_or_fn)
    try:
        loop = asyncio.get_running_loop()
    except RuntimeError:
        loop = None
    if loop is not None and loop.is_running():
        with concurrent.futures.ThreadPoolExecutor(max_workers=1) as pool:
            return pool.submit(lambda: asyncio.run(fn())).result()
    return asyncio.run(fn())


limits, gen_constraints = build_submission_limits()
evaluators = {}
for arm, agent_dir in ARM_DIRS.items():
    evaluators[arm] = Evaluator(EvalConfig(
        tasks_path=TASKS_PATH, snapshots_dir=DATA_DIR / 'snapshots', results_dir=WORKING_DIR / 'results' / arm,
        submission_dir=agent_dir, models=models, sandbox='subprocess',
        timeout_seconds=300, max_time_minutes=20, max_tool_calls=40, max_turns=500,
        limits=limits, generation_constraints=gen_constraints, adapter_manifest=adapters,
        context_cache_config=ContextCacheConfig(min_tokens=2048, ttl_seconds=1800, cache_intervals=10),
        events_compaction_config=EventsCompactionConfig(
            compaction_interval=15, overlap_size=2, token_threshold=14336, event_retention_size=5),
        graph_dir=str(DATA_DIR / 'graphs'), embeddings_dir=str(DATA_DIR / 'embeddings'),
        wheels_dir=DATA_DIR / 'wheels', verbose=False,
    ))

# Alternate the order per task so neither configuration always runs first
rows = []
arm_names = list(ARM_DIRS)
for idx, task in enumerate(SAMPLE_TASKS, start=1):
    order = arm_names if idx % 2 else arm_names[::-1]
    for arm in order:
        t0 = time.time()
        try:
            result = run_sync(evaluators[arm].evaluate_task, task=task, task_index=idx, total_tasks=len(SAMPLE_TASKS))
            rows.append({'arm': arm, 'task': task.instance_id, 'resolved': bool(result.resolved),
                         'tool_calls': result.tool_calls, 'patch_chars': len(result.agent_patch or ''),
                         'error': (result.error or '')[:80], 'seconds': round(result.duration_seconds)})
        except Exception as e:  # keep going so one broken task does not lose the whole run
            rows.append({'arm': arm, 'task': task.instance_id, 'resolved': False,
                         'error': f'{type(e).__name__}: {e}'[:120], 'seconds': round(time.time() - t0)})
        print(rows[-1], flush=True)
        pd.DataFrame(rows).to_csv(WORKING_DIR / 'task_results.csv', index=False)
results_df = pd.DataFrame(rows)
results_df.pivot_table(index='task', columns='arm', values='resolved', aggfunc='first')

# CELL 12: code
import re
from collections import Counter


def edit_stats(trace_path):
    c = Counter()
    prev, streak = None, 0
    for step in json.loads(trace_path.read_text())['steps']:
        obs = (step.get('observation') or {}).get('content', '') or ''
        for call in step.get('tool_calls') or []:
            name, args = call['function_name'], call.get('arguments') or {}
            sig = json.dumps([name, args], sort_keys=True)
            streak = streak + 1 if sig == prev else 1
            prev = sig
            c['max identical calls in a row'] = max(c['max identical calls in a row'], streak)
            c['tool calls'] += 1
            if 'mandatory input parameters are not present' in obs:
                c['calls with lost arguments'] += 1
            if name == 'run_skill_script':
                c['setup ok'] += int('Tests protected' in obs or 'PROTECTED' in obs)
                c['setup by setup step'] += int(call.get('extra', {}).get('author') == 'setup')
            if name == 'edit_file':
                c['edit_file'] += 1
                if 'mandatory input parameters are not present' in obs:
                    c['edit_file: arguments lost'] += 1
                elif '"status": "error"' in obs:
                    c['edit_file: failed'] += 1
                else:
                    c['edit_file: ok'] += 1
            if name == 'run_command' and '.git/agent/edit.py' in (args.get('command') or ''):
                cmd = args.get('command') or ''
                if '--show' in cmd:
                    c['editor: --show'] += 1
                    continue
                c['editor edits'] += 1
                if 'APPLIED to' in obs and 'NOT APPLIED' not in obs:
                    c['editor: applied'] += 1
                    c['editor: applied after repair'] += int('(matched:' in obs)
                else:
                    c['editor: not applied'] += 1
    return c


per_arm, per_task = {}, []
for arm in ARM_DIRS:
    total = Counter()
    for path in sorted((WORKING_DIR / 'results' / arm / 'traces').glob('trace_*.json')):
        s = edit_stats(path)
        per_task.append({'arm': arm, 'task': path.stem.removeprefix('trace_'), **s})
        total.update({k: v for k, v in s.items() if k != 'max identical calls in a row'})
        total['worst identical streak'] = max(total['worst identical streak'], s['max identical calls in a row'])
    sub = results_df[results_df.arm == arm]
    total['tasks'] = len(sub)
    total['resolved'] = int(sub.resolved.sum())
    per_arm[arm] = total

summary = pd.DataFrame(per_arm).fillna(0).astype(int)
display(summary)
display(pd.DataFrame(per_task).fillna(0))
(WORKING_DIR / 'ab_summary.json').write_text(json.dumps({'per_arm': per_arm, 'per_task': per_task}, indent=1, default=int))
