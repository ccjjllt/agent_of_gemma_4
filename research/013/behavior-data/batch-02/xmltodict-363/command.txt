import json
import xmltodict
cases = []
try:
    got = xmltodict.parse('<root> </root>', strip_whitespace=False)
    cases.append({'case': 'whitespace_preserved', 'pass': got == {'root': ' '}, 'got': got})
except Exception as e:
    cases.append({'case': 'whitespace_preserved', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    got = xmltodict.parse('<root> </root>', strip_whitespace=True)
    cases.append({'case': 'whitespace_stripped', 'pass': got == {'root': None}, 'got': got})
except Exception as e:
    cases.append({'case': 'whitespace_stripped', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
print(json.dumps({'cases': cases}, sort_keys=True))
raise SystemExit(0 if cases and all(c['pass'] for c in cases) else 1)
