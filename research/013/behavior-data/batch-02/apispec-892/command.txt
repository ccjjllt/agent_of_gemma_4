import json
import marshmallow
from apispec.ext.marshmallow.field_converter import FieldConverterMixin
calls = []
class C(FieldConverterMixin):
    def __init__(self):
        from packaging.version import Version
        self.openapi_version = Version('3.0.0')
        self.field_mapping = dict(FieldConverterMixin.field_mapping)
conv = C()
try:
    got_ip = conv.field2type_and_format(marshmallow.fields.IPv4())
    calls.append({'case': 'ipv4', 'pass': got_ip == {'type': 'string', 'format': 'ipv4'}, 'got': got_ip})
except Exception as e:
    calls.append({'case': 'ipv4', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    got_ip6 = conv.field2type_and_format(marshmallow.fields.IPv6())
    calls.append({'case': 'ipv6', 'pass': got_ip6 == {'type': 'string', 'format': 'ipv6'}, 'got': got_ip6})
except Exception as e:
    calls.append({'case': 'ipv6', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    got_str = conv.field2type_and_format(marshmallow.fields.String())
    calls.append({'case': 'string_still_works', 'pass': got_str == {'type': 'string'}, 'got': got_str})
except Exception as e:
    calls.append({'case': 'string_still_works', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
print(json.dumps({'cases': calls}, sort_keys=True))
raise SystemExit(0 if calls and all(c['pass'] for c in calls) else 1)
