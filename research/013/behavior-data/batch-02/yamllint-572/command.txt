import contextlib,io,json,os,sys,tempfile
from pathlib import Path
from yamllint import cli
cases=[]
original=Path.cwd()
with tempfile.TemporaryDirectory(prefix='p009-yaml-config-') as tmp:
    root=Path(tmp)
    project=root/'project'
    child=project/'a'/'b'
    child.mkdir(parents=True)
    (project/'.yamllint').write_text('rules: {trailing-spaces: disable}\n')
    explicit=root/'explicit.yml'
    explicit.write_text('rules: {trailing-spaces: disable}\n')
    global_conf=root/'global.yml'
    global_conf.write_text('rules: {trailing-spaces: disable}\n')
    os.environ['YAMLLINT_CONFIG_FILE']=str(global_conf)
    def run(name,cwd,args,expected):
        os.chdir(cwd)
        old_stdin=sys.stdin
        output,error=io.StringIO(),io.StringIO()
        sys.stdin=io.StringIO('key: value  \n')
        try:
            with contextlib.redirect_stdout(output),contextlib.redirect_stderr(error):
                try:
                    cli.run(['-f','parsable','-']+args)
                    code=0
                except SystemExit as e:
                    code=e.code
        finally:
            sys.stdin=old_stdin
            os.chdir(original)
        cases.append({'case':name,'exit':code,'stdout':output.getvalue(),'stderr':error.getvalue(),'pass':code==expected})
    global_conf.write_text('rules: {trailing-spaces: enable}\n')
    run('parent_config',child,[],0)
    (child/'.yamllint').write_text('rules: {trailing-spaces: enable}\n')
    run('current_config_wins',child,[],1)
    run('explicit_config_wins',child,['-c',str(explicit)],0)
    isolated=root/'isolated'
    isolated.mkdir()
    global_conf.write_text('rules: {trailing-spaces: disable}\n')
    run('global_fallback',isolated,[],0)
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
