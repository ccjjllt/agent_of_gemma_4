import json,sys,warnings
warnings.filterwarnings('ignore',category=DeprecationWarning)
from pydantic import BaseModel
from pydantic.color import Color
red=Color('red')
cases=[
 {'case':'same_name','pass':red==Color('red')},
 {'case':'same_rgb','pass':red==Color('#ff0000') and red==Color((255,0,0))},
 {'case':'different_color','pass':red!=Color('blue')},
 {'case':'different_alpha','pass':red!=Color((255,0,0,0.5))},
 {'case':'same_alpha','pass':Color((255,0,0,0.5))==Color('rgba(255,0,0,0.5)')},
 {'case':'non_color','pass':(red=='red') is False and (red==object()) is False},
 {'case':'hash_consistent','pass':len({red,Color('#ff0000')})==1},
]
class Model(BaseModel):
    color:Color
cases.append({'case':'model_equality','pass':Model(color='red')==Model(color='#ff0000')})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
