import copy
import json
from boltons.typeutils import make_sentinel
cases = []
try:
    s = make_sentinel()
    c = copy.copy(s)
    cases.append({'case': 'copy_is_identity', 'pass': c is s, 'same': c is s})
except Exception as e:
    cases.append({'case': 'copy_is_identity', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    s = make_sentinel()
    d = copy.deepcopy(s)
    cases.append({'case': 'deepcopy_is_identity', 'pass': d is s, 'same': d is s})
except Exception as e:
    cases.append({'case': 'deepcopy_is_identity', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
print(json.dumps({'cases': cases}, sort_keys=True))
raise SystemExit(0 if cases and all(c['pass'] for c in cases) else 1)
