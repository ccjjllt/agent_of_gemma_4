import json
from cantools.database.can.formats.arxml.utils import parse_number_string
cases = []
try:
    v = parse_number_string('1e1')
    cases.append({'case': 'sci_int', 'pass': v == 10, 'got': v})
except Exception as e:
    cases.append({'case': 'sci_int', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    v = parse_number_string('1.5e2', allow_float=True)
    cases.append({'case': 'sci_float', 'pass': v == 150.0, 'got': v})
except Exception as e:
    cases.append({'case': 'sci_float', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    v = parse_number_string('0x10')
    cases.append({'case': 'hex_still_works', 'pass': v == 16, 'got': v})
except Exception as e:
    cases.append({'case': 'hex_still_works', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    v = parse_number_string('true')
    cases.append({'case': 'true_still_works', 'pass': v == 1, 'got': v})
except Exception as e:
    cases.append({'case': 'true_still_works', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
print(json.dumps({'cases': cases}, sort_keys=True))
raise SystemExit(0 if cases and all(c['pass'] for c in cases) else 1)
