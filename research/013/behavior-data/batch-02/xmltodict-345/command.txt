import json
import xmltodict
cases = []
try:
    out = xmltodict.unparse({'a': {'@x': '1', '#text': 'hi'}}, full_document=False)
    cases.append({'case': 'attr_unparse', 'pass': out == '<a x="1">hi</a>', 'out': out})
except Exception as e:
    cases.append({'case': 'attr_unparse', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    out = xmltodict.unparse({'a': {'@xmlns': {'': 'urn:x'}, '#text': 'v'}})
    cases.append({'case': 'xmlns_unparse', 'pass': 'xmlns' in out, 'out': out})
except Exception as e:
    cases.append({'case': 'xmlns_unparse', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
try:
    back = xmltodict.parse(xmltodict.unparse({'root': {'item': ['1', '2']}}))
    cases.append({'case': 'roundtrip', 'pass': back == {'root': {'item': ['1', '2']}}, 'out': back})
except Exception as e:
    cases.append({'case': 'roundtrip', 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
print(json.dumps({'cases': cases}, sort_keys=True))
raise SystemExit(0 if cases and all(c['pass'] for c in cases) else 1)
