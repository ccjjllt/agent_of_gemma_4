from apispec.ext.marshmallow.field_converter import FieldConverterMixin
print(json.dumps({'module_origin': FieldConverterMixin.__module__}))
from packaging.version import Version
class C(FieldConverterMixin):
    def __init__(self, v):
        self.openapi_version = Version(v)
        self.field_mapping = dict(FieldConverterMixin.field_mapping)
class S(marshmallow.Schema):

        cases.append({'case': name, 'pass': False, 'error': type(e).__name__ + ': ' + str(e)})
