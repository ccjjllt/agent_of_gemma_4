from pathlib import Path
origin = str(Path('boltons/fileutils.py').resolve())
spec = importlib.util.spec_from_file_location('ws_fileutils', 'boltons/fileutils.py')
mod = importlib.util.module_from_spec(spec); spec.loader.exec_module(mod)

cases = []
def run(name, fn, expect=None, expect_error=None):
    try:
        got = fn()
    except Exception as e:
        msg = type(e).__name__ + ': ' + str(e)
        ok = expect_error is not None and expect_error in msg
        cases.append({'case': name, 'got': 'ERROR:' + msg, 'pass': ok})
        return
    ok = (got == expect) if expect_error is None else False
    cases.append({'case': name, 'got': sorted(got), 'pass': ok})

    run('depth0', lambda: rel(mod.iter_find_files(str(root), '*.txt', max_depth=0)), expect=['top.txt'])
