import json,sys
import numpy as np
from deepdiff import DeepHash
cases=[];hashes=[]
for name,arr in [('true',np.array([True],dtype=bool)),('false',np.array([False],dtype=bool)),('matrix',np.array([[True,False]],dtype=bool)),('int',np.array([1,2],dtype=int))]:
    d={'p':arr}
    try:
        h=DeepHash(d)[d];h2=DeepHash(d)[d]
        hashes.append(h)
        cases.append({'case':name,'pass':isinstance(h,str) and h==h2,'hash':h})
    except Exception as e:
        hashes.append(None);cases.append({'case':name,'pass':False,'error':type(e).__name__+': '+str(e)})
cases.append({'case':'bool_values_distinct','pass':hashes[0] is not None and hashes[0]!=hashes[1]})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
