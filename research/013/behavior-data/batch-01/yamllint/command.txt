import json,sys
from yamllint import linter
from yamllint.config import YamlLintConfig
cases=[]
def check(name,text,opts,count):
    try:
        cfg=YamlLintConfig('rules:\n  quoted-strings:\n'+''.join('    '+k+': '+v+'\n' for k,v in opts.items()))
        errors=list(linter.run(text,cfg))
        cases.append({'case':name,'errors':[str(e) for e in errors],'pass':len(errors)==count})
    except Exception as e:
        cases.append({'case':name,'pass':False,'error':type(e).__name__+': '+str(e)})
opts={'required':'true','quote-type':'double','check-keys':'true'}
check('plain_key','one: 1\n',opts,1)
check('wrong_key_quotes',"'one': 1\n",opts,1)
check('correct_key','"one": 1\n',opts,0)
check('default_key_ignored','one: "v"\n',{'required':'true','quote-type':'double'},0)
check('separate_key_quotes',"'one': \"v\"\n",dict(opts,**{'key-quote-type':'single'}),0)
check('unquoted_value','"one": v\n',opts,1)
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
