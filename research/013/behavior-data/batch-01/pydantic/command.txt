import json, sys
from typing import Annotated
from annotated_types import MaxLen
from pydantic import BaseModel, ValidationError
class Seq(BaseModel):
    inner:list[int]
    def __len__(self): return len(self.inner)
    def __getitem__(self,k): return self.inner[k]
class Model(BaseModel):
    seq:Annotated[Seq,MaxLen(3)]
cases=[]
for n in [2,3,4]:
    try:
        value=Model.model_validate({'seq':{'inner':list(range(n))}})
        accepted=True
    except ValidationError:
        accepted=False
    cases.append({'length':n,'accepted':accepted,'pass':accepted==(n<=3)})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
