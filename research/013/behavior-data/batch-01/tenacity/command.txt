import json,sys
from mypy import api
code="""import tenacity
def stop(s:tenacity.RetryCallState)->bool: return True
def wait(s:tenacity.RetryCallState)->float: return 0.0
def retry(s:tenacity.RetryCallState)->bool: return False
r=tenacity.Retrying(stop=stop,wait=wait,retry=retry)
r.copy(stop=stop,wait=wait,retry=retry)
"""
cases=[]
for name,text,expected in [('callbacks',code,0),('wrong_return',code.replace('def stop(s:tenacity.RetryCallState)->bool: return True','def stop(s:tenacity.RetryCallState)->str: return "bad"'),1)]:
    out,err,status=api.run(['--config-file=/dev/null','--no-incremental','--follow-imports=silent','--ignore-missing-imports','-c',text])
    cases.append({'case':name,'exit':status,'stdout':out,'stderr':err,'pass':status==expected})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
