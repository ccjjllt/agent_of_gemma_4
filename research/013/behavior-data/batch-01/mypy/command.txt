import json,sys
from mypy import api
cases=[]
base="from typing_extensions import TypedDict\nclass A(TypedDict):\n    my_attr_1: str\n    my_attr_2: int\nd: A\n"
for key,expected in [('',1),('my_attr_1',0)]:
    out,err,status=api.run(['--no-incremental','--cache-dir=/tmp/p009-mypy-cache','-c',base+'d['+repr(key)+']\n'])
    ok=status==expected and 'INTERNAL ERROR' not in out+err
    cases.append({'key':key,'exit':status,'stdout':out,'stderr':err[-1000:],'pass':ok})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
