import json,sys
from apispec import APISpec
from apispec.ext.marshmallow import MarshmallowPlugin
from marshmallow import Schema,fields
class S(Schema):
    plain=fields.Dict()
    raw=fields.Dict(values=fields.Raw())
    integer=fields.Dict(values=fields.Integer())
    scalar=fields.Integer()
spec=APISpec(title="API",version="1",openapi_version="3.0.2",plugins=[MarshmallowPlugin()])
spec.components.schema("S",schema=S)
p=spec.to_dict()['components']['schemas']['S']['properties']
cases=[{'case':'plain_dict','schema':p['plain'],'pass':p['plain']=={'type':'object','additionalProperties':{}}},
       {'case':'raw_dict','schema':p['raw'],'pass':p['raw']=={'type':'object','additionalProperties':{}}},
       {'case':'integer_dict','schema':p['integer'],'pass':p['integer']=={'type':'object','additionalProperties':{'type':'integer'}}},
       {'case':'scalar_unchanged','schema':p['scalar'],'pass':p['scalar']=={'type':'integer'}}]
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
