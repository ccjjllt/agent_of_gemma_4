import json,sys
import dataset
db=dataset.connect('sqlite:///:memory:')
table=db['items']
table.insert({'name':'one','old':7})
try:
    table.upsert({'name':'one','new':11},['name'])
    row=table.find_one(name='one')
    ok=row['old']==7 and row['new']==11 and table.count()==1
    result={'pass':ok,'row':dict(row),'columns':list(table.columns)}
except Exception as e:
    ok=False
    result={'pass':False,'error':type(e).__name__+': '+str(e)}
print(json.dumps(result,sort_keys=True))
db.close()
sys.exit(0 if ok else 1)
