import json, sys
from apispec import APISpec
from apispec.ext.marshmallow import MarshmallowPlugin
from marshmallow import Schema, fields
cases=[]
for flag in [True, False, None]:
    meta={'description':'field bar'}
    if flag is not None: meta['deprecated']=flag
    S=type('S',(Schema,),{'bar':fields.Str(required=True,metadata=meta)})
    spec=APISpec(title='API',version='1',openapi_version='3.1.0',plugins=[MarshmallowPlugin()])
    spec.path('/endpoint',operations={'get':{'parameters':[{'in':'query','schema':S}]}})
    p=spec.to_dict()['paths']['/endpoint']['get']['parameters'][0]
    ok=p.get('description')=='field bar' and p.get('required') is True
    ok=ok and (('deprecated' not in p and 'deprecated' not in p['schema']) if flag is None else p.get('deprecated') is flag and 'deprecated' not in p['schema'])
    cases.append({'case':str(flag),'parameter':p,'pass':ok})
class B(Schema):
    bar=fields.Str(metadata={'deprecated':True})
spec.components.schema('Body',schema=B)
prop=spec.to_dict()['components']['schemas']['Body']['properties']['bar']
cases.append({'case':'component_property','pass':prop.get('deprecated') is True})
print(json.dumps({'cases':cases},sort_keys=True))
sys.exit(0 if all(c['pass'] for c in cases) else 1)
